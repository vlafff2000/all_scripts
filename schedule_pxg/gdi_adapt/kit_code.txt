import json
import os
import sys
import time
import traceback
from datetime import datetime

PROJECT_DIR = r"__PROJECT_DIR__"

if not os.path.isdir(os.path.join(PROJECT_DIR, 'pi_dfac')):
    raise RuntimeError('Не найдена папка комплекта: %s (внутри должна быть папка pi_dfac). '
                       'Если комплект перенесли, заново скопируйте код из «Скедул ПХГ».' % PROJECT_DIR)
if PROJECT_DIR not in sys.path:
    sys.path.insert(0, PROJECT_DIR)
for _m in [m for m in sys.modules if m == 'pi_dfac' or m.startswith('pi_dfac.')]:
    del sys.modules[_m]       # при повторном запуске в одной сессии подхватываются правки

from pi_dfac import tnav_api
from pi_dfac.snapshot import save_snapshots

RUN_DIR = os.path.join(PROJECT_DIR, 'runs', datetime.now().strftime('%Y-%m-%d_%H-%M-%S'))
os.makedirs(RUN_DIR)
_log = open(os.path.join(RUN_DIR, 'run.log'), 'w', encoding='utf-8')


def log(s):
    print(s)
    _log.write(s + '\n')
    _log.flush()


try:
    with open(os.path.join(PROJECT_DIR, 'studies.json'), encoding='utf-8') as f:
        studies = json.load(f)
    tnav_api.bind(globals(), 0)
    snaps, failed, t0 = {}, [], time.time()
    for i, s in enumerate(studies, 1):
        try:
            snaps[s['key']] = tnav_api.extract_snapshot(s['well'], datetime.strptime(s['date'], '%Y-%m-%d'))
            log('[%d/%d] %s %s: ок' % (i, len(studies), s['well'], s['date']))
        except Exception as e:      # одна скважина не должна останавливать остальные
            failed.append(s['key'])
            log('[%d/%d] %s %s: ПРОПУЩЕНА - %s' % (i, len(studies), s['well'], s['date'], e))
    save_snapshots(snaps, os.path.join(RUN_DIR, 'snapshots.pkl'))
    log('Готово: прочитано %d из %d, пропущено %d, %.0f с. Файл: %s' % (len(snaps), len(studies), len(failed), time.time() - t0,
                                                                       os.path.join(RUN_DIR, 'snapshots.pkl')))
    log('Вернитесь в «Скедул ПХГ»: вкладка «Адаптация по ГДИ», шаг 5.')
except Exception:
    log('ОШИБКА:\n' + traceback.format_exc())
    _log.close()
    raise
